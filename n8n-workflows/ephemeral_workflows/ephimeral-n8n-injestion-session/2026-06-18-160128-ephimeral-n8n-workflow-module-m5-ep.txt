⏺ // n8n Function node: "Validate"
  // Fail-closed gateway for ephemeral ingestion. Runs before any 
  extract/embed/INSERT.
  const fs = require('fs');
  const path = require('path');

  // n8n varies: payload may be at $json or $json.body
  const src = ($json && $json.body) ? $json.body : $json;

  const conversation_id = src.conversation_id;
  const tenant_id       = src.tenant_id;
  const file_path       = src.file_path;
  const source_name     = src.source_name;
  const rawTtl          = src.ttl_seconds;

  // 1. Required fields (tenant_id is the isolation gate — never proceed without
  it)
  const required = { conversation_id, tenant_id, file_path, source_name };
  for (const [key, val] of Object.entries(required)) {
    if (val === undefined || val === null || (typeof val === 'string' &&
  val.trim() === '')) {
      throw new Error(`Missing required field: ${key}`);
    }
  }

  // 2. Path-traversal guard — resolved path must live under /tmp or /uploads
  const resolved = path.resolve(String(file_path));
  const allowedRoots = ['/tmp/', '/uploads/'];
  const underAllowedRoot = allowedRoots.some(root => resolved.startsWith(root));
  if (!underAllowedRoot) {
    throw new Error(`file_path outside allowed roots (/tmp, /uploads): 
  ${resolved}`);
  }

  // 3. Size cap — 10 MB (MAX_WHATSAPP_UPLOAD_BYTES). statSync also surfaces 
  missing files.
  const MAX_BYTES = 10 * 1024 * 1024; // 10485760
  let stat;
  try {
    stat = fs.statSync(resolved);
  } catch (e) {
    throw new Error(`Cannot stat file_path: ${resolved} (${e.message})`);
  }
  if (!stat.isFile()) {
    throw new Error(`file_path is not a regular file: ${resolved}`);
  }
  if (stat.size > MAX_BYTES) {
    throw new Error(`File exceeds 10 MB limit: ${stat.size} bytes`);
  }

  // 4. TTL bounds — default 3600; if provided must be an integer in [60, 86400]
  let ttl_seconds = 3600;
  if (rawTtl !== undefined && rawTtl !== null && rawTtl !== '') {
    const n = Number(rawTtl);
    if (!Number.isFinite(n) || !Number.isInteger(n)) {
      throw new Error(`ttl_seconds must be an integer: ${rawTtl}`);
    }
    if (n < 60 || n > 86400) {
      throw new Error(`ttl_seconds out of range [60, 86400]: ${n}`);
    }
    ttl_seconds = n;
  }

  return [{
    json: {
      conversation_id: String(conversation_id),
      tenant_id: String(tenant_id),
      file_path: resolved,
      source_name: String(source_name),
      ttl_seconds,
    },
  }];

  6. HTTP status routing

  Maps to 422 (validation):
  - Validate throws — Missing required field…, file_path outside allowed roots…,
  File exceeds 10 MB limit…, ttl_seconds out of range…
  - Extract Text throw — Unsupported file type…

  Maps to 5xx (runtime):
  - Validate throws — Cannot stat file_path…, …is not a regular file… (file
  missing/unreadable → Phase 3 case 5)
  - Extract Text runtime read/parse failure
  - OCR (OpenAI), Voyage embed, Store ephemeral (Postgres) errors

  How the distinction is carried: every throwing node uses On Error = "Continue 
  (using error output)", so its error output emits an item containing the thrown
  error message. All such error outputs fan into the single Format Error node,
  which:
  1. message-matches the known validation prefixes above → sets statusCode = 
  422; otherwise → statusCode = 500;
  2. emits { "status": "failed", "error": "<message>", "statusCode": <code> }.

  Respond (failure) sets its Response Code field to the expression {{ 
  $json.statusCode }} and returns body { "status": "failed", "error": 
  "<message>" }. This is the single place the 422/5xx decision is made, and it
  relies on the stable error-string prefixes already produced by the Phase 4a
  Validate code (I'll keep those strings stable in Phase 4b).

  Node 1 — Extract Text (Function)

  // n8n Function node: "Extract Text"
  // env: NODE_FUNCTION_ALLOW_BUILTIN=fs,path,os,child_process
  //      NODE_FUNCTION_ALLOW_EXTERNAL=pdf-parse,mammoth
  // container: poppler-utils (pdftoppm) required for scanned-PDF OCR (R1)
  const fs = require('fs');
  const path = require('path');
  const os = require('os');
  const { execFileSync } = require('child_process');
  const pdfParse = require('pdf-parse');
  const mammoth = require('mammoth');

  const input = items[0].json;
  const { conversation_id, tenant_id, source_name, ttl_seconds } = input;
  const filePath = input.file_path;

  // Infer type from extension (Validate already guaranteed existence + path 
  safety)
  const ext = path.extname(source_name || filePath).toLowerCase().replace(/^\./,
  '');
  const SUPPORTED = ['pdf', 'docx', 'txt', 'png', 'jpg', 'jpeg'];
  if (!SUPPORTED.includes(ext)) {
    throw new Error(`Unsupported file type: .${ext}`);   // -> 422 (validation)
  }

  const TEXT_THRESHOLD = 20;   // chars; a PDF below this is treated as scanned
  const passthrough = { conversation_id, tenant_id, source_name, ttl_seconds };
  const dataUrl = (buf, mime) =>
  `data:${mime};base64,${buf.toString('base64')}`;
  const pageNum = f => parseInt((f.match(/(\d+)\.png$/) || [])[1] || '0', 10);

  let out;
  try {
    if (ext === 'txt') {
      out = { ...passthrough, source_type: 'txt', text:
  fs.readFileSync(filePath, 'utf8'),
              needs_ocr: false, page_count: 1, images: [] };

    } else if (ext === 'docx') {
      const { value } = await mammoth.extractRawText({ path: filePath });
      out = { ...passthrough, source_type: 'docx', text: value || '',
              needs_ocr: false, page_count: 1, images: [] };

    } else if (ext === 'pdf') {
      const parsed = await pdfParse(fs.readFileSync(filePath));
      const text = (parsed.text || '').trim();
      if (text.length >= TEXT_THRESHOLD) {
        out = { ...passthrough, source_type: 'pdf', text, needs_ocr: false,
                page_count: parsed.numpages || 1, images: [] };
      } else {
        // Scanned PDF -> rasterize each page to PNG, hand images to OCR node 
  (R1)
        const workDir = fs.mkdtempSync(path.join(os.tmpdir(), 'ocr-'));
        execFileSync('pdftoppm', ['-png', '-r', '150', filePath,
  path.join(workDir, 'page')]);
        const files = fs.readdirSync(workDir).filter(f => f.endsWith('.png'))
                        .sort((a, b) => pageNum(a) - pageNum(b));
        const images = files.map(f => ({
          page_number: pageNum(f),
          data_url: dataUrl(fs.readFileSync(path.join(workDir, f)),
  'image/png'),
        }));
        out = { ...passthrough, source_type: 'pdf', text: '', needs_ocr: true,
                page_count: images.length, images };
      }

    } else {
      // png / jpg / jpeg -> single-image OCR
      const mime = ext === 'png' ? 'image/png' : 'image/jpeg';
      out = { ...passthrough, source_type: ext, text: '', needs_ocr: true,
  page_count: 1,
              images: [{ page_number: 1, data_url:
  dataUrl(fs.readFileSync(filePath), mime) }] };
    }
  } catch (e) {
    // read / parse / rasterize failures are runtime -> 5xx
    throw new Error(`Extraction failed for ${source_name}: ${e.message}`);
  }

  return [{ json: out }];

  Node 2 — Chunk (Function)

  // n8n Function node: "Chunk"
  // One input item. Text is either input.text (non-OCR branch) or the OCR HTTP
  // response (OCR node configured with "Include Input Fields"). 512-word 
  windows,
  // 50-word overlap (MODULE_SPEC_M5).
  const input = items[0].json;
  const used_ocr = input.needs_ocr === true;

  let text = (typeof input.text === 'string' && input.text.trim().length)
    ? input.text
    : (input.choices && input.choices[0] && input.choices[0].message
        ? (input.choices[0].message.content || '')
        : '');
  text = String(text).trim();

  if (!text.length) {
    throw new Error(`No extractable text for ${input.source_name}`);   // -> 5xx
  (runtime)
  }

  const CHUNK_SIZE = 512;   // words
  const OVERLAP = 50;
  const words = text.split(/\s+/).filter(Boolean);
  const chunks = [];
  let i = 0;
  while (i < words.length) {
    chunks.push({
      content: words.slice(i, i + CHUNK_SIZE).join(' '),
      chunk_index: chunks.length,
      page_number: null,     // page boundaries are lost after concat; 
  metadata.page_number optional
    });
    i += (CHUNK_SIZE - OVERLAP);
  }

  return [{
    json: {
      chunks,
      conversation_id: input.conversation_id,
      tenant_id: input.tenant_id,
      source_name: input.source_name,
      ttl_seconds: input.ttl_seconds,
      used_ocr,

  const input = items[0].json;

      },
    };
  });

    'Unsupported file type',
  ];
  const isValidation = VALIDATION_MARKERS.some(m => String(msg).includes(m));

  return [{
    json: {
      status: 'failed',
      error: String(msg),
      statusCode: isValidation ? 422 : 500,
    },
  }];
